CMAKE_MINIMUM_REQUIRED(VERSION 3.22)

PROJECT(UEFIPatch LANGUAGES C CXX)

SET(CMAKE_CXX_STANDARD 11)
SET(CMAKE_CXX_STANDARD_REQUIRED ON)
SET(CMAKE_CXX_EXTENSIONS OFF)

INCLUDE(${CMAKE_CURRENT_SOURCE_DIR}/../common/edit_sources.cmake)

ADD_DEFINITIONS(
 -DU_ENABLE_NVRAM_PARSING_SUPPORT
 -DU_ENABLE_ME_PARSING_SUPPORT
 -DU_ENABLE_FIT_PARSING_SUPPORT
 -DU_ENABLE_GUID_DATABASE_SUPPORT
)

ADD_EXECUTABLE(UEFIPatch uefipatch_main.cpp uefipatch.cpp ${UEFI_EDIT_COMMON_SOURCES})

IF(UNIX)
 SET_TARGET_PROPERTIES(UEFIPatch PROPERTIES OUTPUT_NAME uefipatch)
ENDIF()

INSTALL(TARGETS UEFIPatch RUNTIME DESTINATION bin)
